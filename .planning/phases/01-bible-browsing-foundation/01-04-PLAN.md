```typescript
const handleVerseClick = useCallback((verseNumber: string) => {
  if (!selectedBook || !selectedChapter || !chapterContent) return;

  // Build passage ID for single verse
  const passageId = `${selectedBook.id}.${selectedChapter.number}.${verseNumber}`;

  setIsLoadingContent(true);
  setError(null);

  bibleApi.getPassage(selectedBibleId, passageId)
    .then((passage) => {
      setPassageOverride({
        reference: passage.reference,
        content: passage.content,
      });
    })
    .catch((err) => {
      console.error('Error loading verse:', err);
      setError('Could not load verse. Try again.');
    })
    .finally(() => {
      setIsLoadingContent(false);
    });
}, [selectedBook, selectedChapter, chapterContent, selectedBibleId]);
```
